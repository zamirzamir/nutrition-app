SELECT email, paid_at, expires_at FROM subscriptions ORDER BY paid_at DESC LIMIT 20;

-- ШАГ 2. Продлить клиенту: подставь его email из шага 1 вместо CLIENT@EMAIL
UPDATE subscriptions
SET expires_at = paid_at + Interval("P95D"),
    updated_at = CurrentUtcTimestamp()
WHERE email = 'CLIENT@EMAIL';
